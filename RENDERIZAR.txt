Comandos para renderizar os arquivos XML e gerar o livro BLFS em vários formatos


1 - Para a versão SystemV do livro:

( time LC_ALL=C make -j1 V=1 L=pt_BR REV=sysv BASEDIR="${HOME}"/sysv nochunks 2>&1 | tee "${HOME}"/genhtml-nochunks-sysv.log.txt && exit "${PIPESTATUS}" )
mv -v BLFS-BOOK.html BLFS-SYSV-BOOK-12.1-pt_BR-NOCHUNKS.html
lynx -dump BLFS-SYSV-BOOK-12.1-pt_BR-NOCHUNKS.html 1> BLFS-SYSV-BOOK-12.1-pt_BR-NOCHUNKS.txt
cd ..
rm -rf 1/
mkdir -v /tmp/1; cd ${_}; ls -al
make BASEDIR=/tmp/1 pdf
make BASEDIR=/tmp/1
tar --create --file="${HOME}"/repos/BLFS/BLFS-SYSV-BOOK-12.1-pt_BR-HTML.tar --wildcards .
xz --verbose "${HOME}"/repos/BLFS/BLFS-SYSV-BOOK-12.1-pt_BR-HTML.tar



2 - Para a versão systemd do livro:

export REV="systemd"
echo ${REV}
mkdir -v /tmp/1; cd "${_}"; ls -al
make BASEDIR=/tmp/1 nochunks
mv -v BLFS-SYSD-BOOK.html BLFS-SYSD-BOOK-12.1-pt_BR-NOCHUNKS.html
lynx -dump BLFS-SYSD-BOOK-12.1-pt_BR-NOCHUNKS.html 1> BLFS-SYSD-BOOK-12.1-pt_BR-NOCHUNKS.txt
cd ..
rm -rf 1/
mkdir -v /tmp/1; cd "${_}"; ls -al
make BASEDIR=/tmp/1 pdf
mv -v BLFS-SYSD-BOOK.pdf BLFS-SYSD-BOOK-12.1-pt_BR.pdf
make BASEDIR=/tmp/1
tar --create --file="${HOME}"/repos/BLFS/BLFS-SYSD-BOOK-12.1-pt_BR-HTML.tar --wildcards .
xz --verbose "${HOME}"/repos/BLFS/BLFS-SYSD-BOOK-12.1-pt_BR-HTML.tar.bz2
